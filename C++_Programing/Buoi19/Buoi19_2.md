# Thuật Toán Sắp Xếp Trộn (Merge Sort). Đếm Số Cặp Nghịch Thế Trong Mảng Bằng Merge Sort.

## 1. Thuật toán sắp xếp trộn (Merge Sort)

- Tư tưởng : Thuật toán sắp xếp trộn sử  dụng phương pháp chia và trị, chia dãy ban đầu thành các dãy con cho tới khi dãy chỉ còn 1 phần tử, sau đó thực hiện trộn 2 dãy con tăng dần thành 1 dãy tăng dần với độ phức tạp tuyến tính 

- Độ phức tạp : $O(n log n)$

- Lưu ý : Dù trường hợp tệ nhất vẫn là $O(n log n)$ nhưng Tốn nhiều bộ nhớ hơn các thuật toán khác.

- Ví dụ : Cho mảng `arr = [6,5,12,10,9,1]`

```text
                                 ┌───┬───┬────┬────┬───┬───┐
```

- Mã giả : 

```text
MergeSort(A, l, r):
    if l >= r
        return
    m = (l+r)/2
    // chia đôi mảng
    mergeSort(A, l, m)
    mergeSort(A, m+1, r)
    // trộn 2 mảng
    merge(A, l, m, r)
```

  - Trộn 2 mảng đã được sắp xếp

- Code thuật toán sẵp xếp trộn 
```cpp

// Tron 2 day con : day 1 [l,m], day 2 [m+1,r]
void merge(int a[], int l, int m, int r){
    // copy mang [l,m] vo vector x
    vector<int> x(a+l,a+m+1);
    // copy mang [m+1,r] vo vector y
    vector<int> y(a + m + 1, a + r + 1);
    // khoi tao 2 bien con tro 
    int i =0, j = 0;
    while(i < x.size() && j <y.size()){
        // neu phan tu x[i] nho hon
        // them phan tu vo mang ket qua 
        if(x[i] <= y[j]){
            // gan bien x cho phan tu a[l]
            a[l] = x[i];
            l++; i++;
        }else{
            // neu th nguoc lai 
            a[l] = y[j];
            l++; j++;
        }
        // neu mang co phan tu con lai 
        while( i < x.size()){
            a[l] = x[i];
            l++; i++;
        }
        // neu mang x het, y con 
        while( j < y.size()){
            a[l] = y[j];
            l++; j++;
        }
    }
}

// thuat toan merge sort
void mergesort(int a[], int l, int r){
    // khoi tao buoc dung 
    if(l >= r) return;
    // tinh phan tu o giua
    int m = (l + r) /2;
    // chia mang [l,m]
    mergesort(a,l,m);
    // chia mang [m+1,r]
    mergesort(a,m+1,r);
    // tron lai 2 mang 
    merge(a,l,m,r);
}
```

## 2. Đếm số cặp nghịch thế trong một mảng (sử dụng mergesort)

- Số nghịch thế  là số cặp (i, j) sao cho i < j và a[i] > a[j]
- Ví dụ : Cho mảng a = [8,7,1,9,2,3]

### Cách 1 : Phương pháp $O(n^2)$
```cpp
#include <iostream>
using namespace std;

//Ham tinh so cap nghich the trong mang
int count_inversions(int a[], int n){
    //Khoi tao bien dem so cap nghich the
    int count = 0;
    //Duyet qua mang
    for(int i = 0; i<n-1; i++){
        //Duyet qua cac phan tu sau i
        for(int j = i + 1; j<n; j++){
            //Neu tim thay cap nghich the thi tang bien dem
            if(a[i] > a[j]){
                count++;
            }
        }
    }
    //Tra ve so cap nghich the
    return count;
}

int main(){
    //Khoi tao mang
    int a[] = {8, 7, 1, 9, 2, 3};
    //Tinh chieu dai mang
    int n = sizeof(a) / sizeof(a[0]);
    //In ra so cap nghich the
    cout << "So cap nghich the: " << count_inversions(a, n) << endl;
    return 0;
}
```

### Cách 2 : Phương pháp $O(nlogn)$

- Sử dụng mergesort để đếm số cặp nghịch thế
- Ý tưởng : 
  - Nếu phần tử của mảng con trái lớn hơn phần tử của mảng con phải thì số cặp nghịch thế bằng số phần tử còn lại của mảng con trái.
  
- Code : 
```cpp
#include <bits/stdc++.h>
using namespace std;

// thuat toan tron 2 mang 
int merge(int a[], int l, int r){
    vector<int> x(a, l, m);
    vector<int> y(a,m+1,r);
    // khoi tao 2 bien i, j 
    int i =0, j =0;
    // khoi tao bien dem 
    int cnt = 0;
    while(i < x.size() && j < y.size()){
        // neu x nho hon y thi them phan tu vo
        // vi pham cap nghich the
        if(x[i] <= y[j]){
            a[l] = x[i];
            l++; i++;
        // day la cap nghich the
        // so cap nghich the thi so phan tu con lai cua x[i] deu lon hon y[j]
        }else{
            cnt += x.size() - i;
            a[l] = y[j];
            j++; l++;
        }
    }
    // them cac phan tu con lai 
    while(i < x.size()){
        a[l] = x[i];
        l++; i++;
    }
    while(j < y.size()){
        a[l] = y[j];
        l++; j++;
    }
    return cnt;
}

// thuat toan tim so cap nghich the 
int count_inversions(int a[], int l, int r){
    // dieu kien dung 
    if(l == r) return 0;
    // chia mang lam 2 
    int m = (l+r) / 2;
    // de quy tim so cap nghich the 
    int cnt = 0;
    cnt += count_inversions(a, l, m);
    cnt += count_inversions(a, m+1, r);
    // tron 2 mang 
    cnt += merge(a, l, m, r);
    return cnt; 
}

int main(){
    int a[] = {8, 7, 1, 9, 2, 3};
    int n = sizeof(a) / sizeof(a[0]);
    cout << "So cap nghich the: " << count_inversions(a, 0, n-1) << endl;
    return 0;
}
```
